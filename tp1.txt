package com.example.proyecto.runner;

import com.example.marketingdigital.*;
import com.example.comunidadvg.*;
import org.springframework.boot.CommandLineRunner;
import org.springframework.stereotype.Component;
import java.time.LocalDate;
import java.time.LocalDateTime;

@Component
public class AcademiaRunner implements CommandLineRunner {

    @Override
    public void run(String... args) throws Exception {
        System.out.println("=== INICIO PRUEBA: SISTEMA DE CAMPAÑAS DE MARKETING ===");
        
        // 1. Crear plataformas
        Plataforma googleAds = new Plataforma(1L, "Google Ads", "https://ads.google.com");
        Plataforma metaAds = new Plataforma(2L, "Meta Ads", "https://business.facebook.com");
        
        // 2. Crear campaña
        Campaña campañaVerano = new Campaña(1L, "Promoción Verano 2026", 50000.0, LocalDate.of(2026, 11, 1));
        
        // 3. Registrar conversiones
        Conversión conv1 = new Conversión(1L, "venta", 150.0, 
                LocalDateTime.of(2026, 11, 2, 14, 30), campañaVerano, googleAds);
        Conversión conv2 = new Conversión(2L, "registro", 0.0, 
                LocalDateTime.of(2026, 11, 2, 16, 15), campañaVerano, metaAds);
        
        campañaVerano.agregarConversion(conv1);
        campañaVerano.agregarConversion(conv2);
        
        // Mostrar datos
        System.out.println("Campaña: " + campañaVerano.getNombreCampana());
        System.out.println("Presupuesto: $" + campañaVerano.getPresupuesto());
        System.out.println("Conversiones registradas: " + campañaVerano.getConversiones().size());
        campañaVerano.getConversiones().forEach(c -> 
                System.out.println("  - " + c.getTipoConversion() + " en " + c.getPlataforma().getNombrePlataforma()
                        + " | Valor: $" + c.getValor()));

        System.out.println("\n=== INICIO PRUEBA: COMUNIDAD DE VIDEOJUEGOS ===");
        
        // 1. Crear usuario
        Usuario usuario1 = new Usuario(1L, "GamerTucuman", LocalDate.of(2026, 3, 15));
        
        // 2. Crear publicación
        Publicacion post1 = new Publicacion(1L, "¡Alguien para jugar en equipo esta noche!",
                LocalDateTime.of(2026, 9, 10, 20, 45), usuario1);
        usuario1.crearPublicacion(post1);
        
        // 3. Otro usuario reacciona
        Usuario usuario2 = new Usuario(2L, "ArnedoJ", LocalDate.of(2026, 5, 20));
        Reaccion reaccion1 = new Reaccion(1L, "corazón", LocalDateTime.of(2026, 9, 11, 8, 10), usuario2, post1);
        usuario2.reaccionar(reaccion1);
        post1.agregarReaccion(reaccion1);
        
        // Mostrar datos
        System.out.println("Usuario: " + usuario1.getNombreUsuario());
        System.out.println("Publicación: " + post1.getContenido());
        System.out.println("Reacciones: " + post1.getReacciones().size());
        post1.getReacciones().forEach(r ->
                System.out.println("  - " + r.getUsuario().getNombreUsuario() + " le dio " 
                        + r.getTipoReaccion() + " el " + r.getFechaReaccion().toLocalDate()));

        System.out.println("\n✅ Todo funcionó correctamente. Fin de la ejecución.");
    }
}
📝 EXPLICACIÓN DE LO HECHO

📌 Parte 1 — Campañas de Marketing

* Modelamos las tres entidades tal cual pide el enunciado: Campaña, Plataforma y Conversión.

* La clase Conversión actúa como tabla intermedia porque vincula cada evento con su campaña y su plataforma de origen. Así podemos saber exactamente qué plataforma generó cada resultado.

* Cada conversión tiene su propio valor y tipo, lo que permite calcular el retorno de inversión por plataforma.

📌 Parte 2 — Comunidad de Videojuegos

* La relación entre usuarios y publicaciones es sencilla: un usuario es autor de sus posts.

* Las reacciones necesitan una tabla intermedia porque es una relación muchos a muchos: un usuario puede reaccionar a muchas publicaciones, y una publicación puede recibir reacciones de muchos usuarios. Por eso creamos la clase Reaccion con referencias a ambos lados.

📌 Sobre la estructura del proyecto

* El código está organizado en paquetes separados para cada sistema: marketingdigital y comunidadvg.

* La clase AcademiaRunner está en el paquete runner como indicó el profesor.

* Usamos java.time para las fechas que es lo recomendado en versiones actuales de Java.
